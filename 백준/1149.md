# RGB 거리
* 시간 : 20분
* 풀이 방법을 몰라서 풀이 방법 확인
* 답지는 보지 않음

```java
import java.io.*;
import java.util.*;

public class Main {
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

    public static void main(String[] args) throws IOException {
        StringTokenizer st = new StringTokenizer(br.readLine());
        int n = Integer.parseInt(st.nextToken());
        int[][] price = new int[n + 1][3];
        int[][] dp = new int[n + 1][3];
        for (int i = 1; i <= n; i++) {
            st = new StringTokenizer(br.readLine());
            price[i][0] = Integer.parseInt(st.nextToken());
            price[i][1] = Integer.parseInt(st.nextToken());
            price[i][2] = Integer.parseInt(st.nextToken());
        }
        dp[0][0] = 0;
        dp[0][1] = 0;
        dp[0][2] = 0;
        dp[1][0] = price[1][0];
        dp[1][1] = price[1][1];
        dp[1][2] = price[1][2];

        if(n < 2){
            System.out.println(Math.min(dp[1][2],Math.min(dp[1][1],dp[1][0])));
        }else {
            for (int i = 2; i <= n; i++) {
                dp[i][0] = Math.min(dp[i-1][1], dp[i-1][2]) + price[i][0];
                dp[i][1] = Math.min(dp[i-1][0], dp[i-1][2]) + price[i][1];
                dp[i][2] = Math.min(dp[i-1][0], dp[i-1][1]) + price[i][2];
            }

            System.out.println(Math.min(dp[n][2],Math.min(dp[n][1],dp[n][0])));
        }
    }
}
```
