# 유기농 배추
* 시간 : 10분

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
    private static boolean[][] graph;

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int T = Integer.parseInt(br.readLine());

        StringTokenizer st;
        for (int i = 0; i < T; i++) {
            st = new StringTokenizer(br.readLine());
            int M = Integer.parseInt(st.nextToken());   //가로
            int N = Integer.parseInt(st.nextToken());   //세로
            int K = Integer.parseInt(st.nextToken());   //배추 위치 개수
            graph = new boolean[N][M];
            for (int j = 0; j < K; j++) {
                st = new StringTokenizer(br.readLine());
                int x = Integer.parseInt(st.nextToken());
                int y = Integer.parseInt(st.nextToken());
                graph[y][x] = true;
            }
            int area = 0;
            for (int j = 0; j < N; j++) {
                for (int k = 0; k < M; k++) {
                    if (graph[j][k]) {
                        area++;
                        bfs(k, j);
                    }
                }
            }
            System.out.println(area);
        }
    }

    private static void bfs(int x, int y) {
        Queue<Coordinate> sequence = new LinkedList<>();
        sequence.add(new Coordinate(x, y));

        int[] dx = {0, 0, 1, -1};
        int[] dy = {1, -1, 0, 0};

        while (!sequence.isEmpty()) {
            Coordinate current = sequence.poll();
            for (int k = 0; k < 4; k++) {
                int nx = current.x + dx[k];
                int ny = current.y + dy[k];
                if (nx >= 0 && graph[0].length > nx && ny >= 0 && ny < graph.length) {
                    if (graph[ny][nx]) {
                        sequence.add(new Coordinate(nx, ny));
                        graph[ny][nx] = false;
                    }
                }
            }
        }
    }

    private static class Coordinate {
        int x;
        int y;

        public Coordinate(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }


}
```
