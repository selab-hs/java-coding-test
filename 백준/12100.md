```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int N = Integer.parseInt(st.nextToken());
        int[][] graph = new int[N][N];
        for (int i = 0; i < N; i++) {
            st = new StringTokenizer(br.readLine());
            for (int j = 0; j < N; j++) {
                graph[i][j] = Integer.parseInt(st.nextToken());
            }
        }

        dfs(graph, 0);

        System.out.println(max);
    }

    private static int max = 0;

    public static void dfs(int[][] graph, int chance) {
        if (chance == 5) {
            max = Math.max(max, getMax(graph));
        }else {
            dfs(tpUp(graph), chance + 1);
            dfs(toRight(graph), chance + 1);
            dfs(toLeft(graph), chance + 1);
            dfs(tpDown(graph), chance + 1);
        }
    }

    public static int[][] tpUp(int[][] graph) {
        int[][] upGraph = new int[graph.length][graph.length];
        for (int i = 0; i < graph.length; i++) {
            Stack<Integer> stack = new Stack<>();
            for (int j = 0; j < graph[i].length; j++) {
                if (graph[j][i] == 0) continue; //0이면 빈칸이므로 넘김

                if (stack.isEmpty()) {
                    stack.push(graph[j][i]);
                    continue;
                }

                if (graph[j][i] == stack.peek()) {
                    stack.pop();
                    stack.push(graph[j][i] * 2);
                } else {
                    stack.push(graph[j][i]);
                }
            }


            for (int j = stack.size() - 1; j >= 0; j--) {
                if (!stack.isEmpty()) {
                    upGraph[j][i] = stack.pop();
                } else {
                    upGraph[j][i] = 0;
                }
            }
        }
        return upGraph;
    }


    public static int[][] tpDown(int[][] graph) {
        int[][] downGraph = new int[graph.length][graph.length];
        for (int i = graph.length - 1; i >= 0; i--) {
            Stack<Integer> stack = new Stack<>();
            for (int j = graph[i].length - 1; j >= 0; j--) {
                if (graph[j][i] == 0) continue; //0이면 빈칸이므로 넘김

                if (stack.isEmpty()) {
                    stack.push(graph[j][i]);
                    continue;
                }

                if (graph[j][i] == stack.peek()) {
                    stack.pop();
                    stack.push(graph[j][i] * 2);
                } else {
                    stack.push(graph[j][i]);
                }
            }

            List<Integer> list = new ArrayList<>(stack);
            for (int j = downGraph.length - 1; j >= 0; j--) {
                if (list.isEmpty()) {
                    downGraph[j][i] = 0;
                } else {
                    downGraph[j][i] = list.get(0);
                    list.remove(0);
                }
            }
        }
        return downGraph;
    }

    //왼쪽
    public static int[][] toLeft(int[][] graph) {
        int[][] leftGraph = new int[graph.length][graph.length];
        for (int i = 0; i < graph.length; i++) {
            Stack<Integer> stack = new Stack<>();
            for (int j = 0; j < graph.length; j++) {
                if (graph[i][j] == 0) continue;
                if (stack.isEmpty()) {
                    stack.push(graph[i][j]);
                    continue;
                }
                if (graph[i][j] == stack.peek()) {
                    stack.pop();
                    stack.push(graph[i][j] * 2);
                } else {
                    stack.push(graph[i][j]);
                }
            }

            for (int j = stack.size() - 1; j >= 0; j--) {
                if (!stack.isEmpty()) {
                    leftGraph[i][j] = stack.pop();
                } else {
                    leftGraph[i][j] = 0;
                }
            }
        }

        return leftGraph;
    }

    //오른쪽
    public static int[][] toRight(int[][] graph) {
        int[][] rightGraph = new int[graph.length][graph.length];
        for (int i = graph.length - 1; i >= 0; i--) {
            Stack<Integer> stack = new Stack<>();
            for (int j = graph.length - 1; j >= 0; j--) {
                if (graph[i][j] == 0) continue;
                if (stack.isEmpty()) {
                    stack.push(graph[i][j]);
                    continue;
                }
                if (graph[i][j] == stack.peek()) {
                    stack.pop();
                    stack.push(graph[i][j] * 2);
                } else {
                    stack.push(graph[i][j]);
                }
            }

            List<Integer> list = new ArrayList<>(stack);
            for (int j = rightGraph.length - 1; j >= 0; j--) {
                if (list.isEmpty()) {
                    rightGraph[i][j] = 0;
                } else {
                    rightGraph[i][j] = list.get(0);
                    list.remove(0);
                }
            }
        }

        return rightGraph;
    }

    private static int getMax(int[][] graph) {
        IntSummaryStatistics statistics = Arrays.stream(graph)
                .parallel()
                .flatMapToInt(Arrays::stream)
                .summaryStatistics();
        return statistics.getMax();
    }
}
```

```java

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class p12100 {
    
    static int n, answer, map[][];
    
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        n = Integer.parseInt(br.readLine());
        answer = 0;
        map = new int[n][n];
        StringTokenizer stz;
        for(int i = 0; i < n; i++) {
            stz = new StringTokenizer(br.readLine());
            for(int j = 0; j < n; j++)
                map[i][j] = Integer.parseInt(stz.nextToken());
        }
        
        game(0);
        System.out.println(answer);
    }
    
    public static void game(int count) {
        if(count == 5) {
            findMax();
            return;
        }
        int copy[][] = new int[n][n];
        for(int i = 0; i < n; i++)
            copy[i] = map[i].clone();
        
        for(int i = 0; i < 4; i++) {
            move(i);
            game(count+1);
            for(int a = 0; a < n; a++)
                map[a] = copy[a].clone();
        }
    }
    
    public static void move(int dir) {
        switch(dir) {
            //위로 몰기
            case 0:
                for(int i = 0; i < n; i++) {
                    int index = 0;
                    int block = 0;
                    for(int j = 0; j < n; j++) {
                        if(map[j][i] != 0) {
                            if(block == map[j][i]) {
                                map[index - 1][i] = block * 2;
                                block = 0;
                                map[j][i] = 0;
                            }
                            else {
                                block = map[j][i];
                                map[j][i] = 0;
                                map[index][i] = block;
                                index++;
                            }
                        }
                    }
                }
                break;
            //왼쪽으로 몰기
            case 1:
                for(int i = 0; i < n; i++) {
                    int index = n - 1;
                    int block = 0;
                    for(int j = n - 1; j >= 0; j--) {
                        if(map[j][i] != 0) {
                            if(block == map[j][i]) {
                                map[index + 1][i] = block * 2;
                                block = 0;
                                map[j][i] = 0;
                            }
                            else {
                                block = map[j][i];
                                map[j][i] = 0;
                                map[index][i] = block;
                                index--;
                            }
                        }
                    }
                }
                break;
            //왼쪽으로 몰기
            case 2:
                for(int i = 0; i < n; i++) {
                    int index = 0;
                    int block = 0;
                    for(int j = 0; j < n; j++) {
                        if(map[i][j] != 0) {
                            if(block == map[i][j]) {
                                map[i][index - 1] = block * 2;
                                block = 0;
                                map[i][j] = 0;
                            }
                            else {
                                block = map[i][j];
                                map[i][j] = 0;
                                map[i][index] = block;
                                index++;
                            }
                        }
                    }
                }
                break;
            //오른쪽으로 몰기
            case 3:
                for(int i = 0; i < n; i++) {
                    int index = n - 1;
                    int block = 0;
                    for(int j = n - 1; j >= 0; j--) {
                        if(map[i][j] != 0) {
                            if(block == map[i][j]) {
                                map[i][index + 1] = block * 2;
                                block = 0;
                                map[i][j] = 0;
                            }
                            else {
                                block = map[i][j];
                                map[i][j] = 0;
                                map[i][index] = block;
                                index--;
                            }
                        }
                    }
                }
                break;
        }
    }
    
    public static void findMax() {
        for(int i = 0; i < n; i++)
            for(int j = 0; j < n; j++)
                answer = Math.max(answer, map[i][j]);
    }
}
```
