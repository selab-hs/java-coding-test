```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        List<Line> lines = new ArrayList<>();
        for (int i = 0; i < 3; i++) {
            StringTokenizer st = new StringTokenizer(br.readLine());
            Line line = new Line(Integer.parseInt(st.nextToken()), Integer.parseInt(st.nextToken()));
            lines.add(line);
        }
        Set<Integer> xVertex = new HashSet<>();
        Set<Integer> yVertex = new HashSet<>();
        for (int i = 0; i < 3; i++) {
            Line line = lines.get(i);
            if (xVertex.contains(line.x)) {
                xVertex.remove(line.x);
            } else {
                xVertex.add(line.x);
            }
            if (yVertex.contains(line.y)) {
                yVertex.remove(line.y);
            } else {
                yVertex.add(line.y);
            }
        }

        Iterator<Integer> iterator = xVertex.iterator();
        Iterator<Integer> iterator2 = yVertex.iterator();
        int x = iterator.next();
        int y = iterator2.next();
        System.out.println(x + " " + y);
    }

    static class Line {
        final int x;
        final int y;

        public Line(int x, int y) {
            this.x = x;
            this.y = y;
        }
    }
}
```
