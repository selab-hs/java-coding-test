# 여행 가자
* 풀이 확인

```java
import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.StringTokenizer;

public class Main {
    static int[] parent;

    public static void main(String[] args) throws NumberFormatException, IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
        StringTokenizer st;

        int N = Integer.parseInt(br.readLine());    //도시 수
        int M = Integer.parseInt(br.readLine());    //여행 일정 길이

        parent = new int[N + 1];
        for (int i = 1; i <= N; i++) {
            parent[i] = i;
        }

        //아어질 수 있는 도시를 입력 받는 반복문
        for (int i = 1; i <= N; i++) {  //parent가 N+1로 생성됐기 때문에 1부터 시작
            st = new StringTokenizer(br.readLine());
            for (int j = 1; j <= N; j++) {
                int temp = Integer.parseInt(st.nextToken());    //여행 가능한지 0 or 1로 받음

                // 연결된 부분은 합집합 연산함.
                if (temp == 1) {    //만약 이동 가능한 도시면
                    union(i, j);    //union 메소드 호출
                }
            }
        }

        st = new StringTokenizer(br.readLine());
        int start = find(Integer.parseInt(st.nextToken())); //여행 시작점
        for (int i = 1; i < M; i++) {
            int now = Integer.parseInt(st.nextToken()); //지금 어디에 가야하는지

            // 맨 처음 출발 도시와 연결되어있지 않은 도시가 있으면
            // 여행 계획이 불가능한 것임.
            if (start != find(now)) {   //지금 있는 곳이 같은 그룹에 속하지 않으면 연산 종료
                bw.write("NO\n");
                bw.flush();
                bw.close();
                br.close();
                return;
            }
        }

        bw.write("YES\n");
        bw.flush();
        bw.close();
        br.close();
    }

    // 이 연산에 들어왔다는 것은 연결되어있다는 의미이므로 같은 그룹임을 의미한다.
    // y의 부모를 x의 부모로 치환하는 연산 (x > y 일 경우, 반대)
    public static void union(int x, int y) {
        x = find(x);    //X의 부모를 찾는다
        y = find(y);    //Y의 부모를 찾는다

        if (x != y) {   //만약 부모가 다르다면 더 작은 수로 바꿔준다
            if (x < y) {
                parent[y] = x;
            } else {
                parent[x] = y;
            }
        }
    }

    // x의 부모를 찾는 연산
    public static int find(int x) {
        if (x == parent[x]) {
            return x;
        }

        return parent[x] = find(parent[x]); //끝까지 찾야한다
    }
}
```
