# 바이러스

* 풀이 시간 : 10분

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int numberOfComputer = Integer.parseInt(br.readLine());
        int numberOfVertex = Integer.parseInt(br.readLine());
        boolean[][] graph = new boolean[numberOfComputer + 1][numberOfComputer + 1];
        StringTokenizer st;
        for (int i = 0; i < numberOfVertex; i++) {
            st = new StringTokenizer(br.readLine());
            int a = Integer.parseInt(st.nextToken());
            int b = Integer.parseInt(st.nextToken());
            graph[a][b] = true;
            graph[b][a] = true;
        }
        System.out.println(bfs(graph, new boolean[numberOfComputer + 1]));
    }

    private static int bfs(boolean[][] graph, boolean[] visited) {
        Set<Integer> union = new HashSet<>();
        Queue<Integer> sequence = new LinkedList<>();
        visited[1] = true;
        sequence.add(1);
        while (!sequence.isEmpty()) {
            int node = sequence.poll();
            for (int i = 1; i < graph[node].length; i++) {
                if (graph[node][i] && !visited[i]) {
                    visited[i] = true;
                    sequence.add(i);
                }
            }
            union.add(node);
        }
        return union.size() - 1;
    }
}
```
